function walk(dir, out = []) {
  if (!fs.existsSync(dir)) return out;

  for (const item of fs.readdirSync(dir)) {
    const full = path.join(dir, item);
    const stat = fs.statSync(full);

    if (stat.isDirectory()) {
      if (!["node_modules", ".next", ".git"].includes(item)) {
        walk(full, out);
      }
    } else if (/\.(tsx|ts|css)$/.test(item)) {
      out.push(full);
    }
  }

  return out;
}

function backup(file) {
  fs.copyFileSync(file, `${file}.before-clean-orange-${stamp}`);
}

// @ts-nocheck

import { NextResponse } from "next/server";
import { createClient } from "@supabase/supabase-js";
import { mkdir, writeFile } from "fs/promises";
import path from "path";

export const runtime = "nodejs";
export const dynamic = "force-dynamic";

function getSupabaseAdmin() {
  const url = process.env.NEXT_PUBLIC_SUPABASE_URL || process.env.SUPABASE_URL || "";
  const key = process.env.SUPABASE_SERVICE_ROLE_KEY || process.env.NEXT_PUBLIC_SUPABASE_ANON_KEY || "";

  if (!url || !key) return null;

  return createClient(url, key, {
    auth: {
      persistSession: false,
      autoRefreshToken: false,
    },
  });
}

function clean(value: FormDataEntryValue | null) {
  return String(value || "").trim();
}

function safeFileName(name: string) {
  return name
    .normalize("NFD")
    .replace(/[\u0300-\u036f]/g, "")
    .replace(/[^a-zA-Z0-9._-]/g, "-")
    .toLowerCase();
}

export async function POST(request: Request) {
  const form = await request.formData();

  const client_email = clean(form.get("client_email")).toLowerCase();
  const client_name = clean(form.get("client_name"));
  const package_name = clean(form.get("package_name")) || "New Mexico Standard";
  const amount = Number(clean(form.get("amount")) || 0);
  const reference = clean(form.get("reference"));
  const phone = clean(form.get("phone"));
  const notes = clean(form.get("notes"));
  const lang = clean(form.get("lang")) || "fr";
  const file = form.get("proof_file") as File | null;

  if (!client_email || !client_email.includes("@")) {
    return NextResponse.redirect(new URL("/fr/virement?error=email", request.url), 303);
  }

  let proof_url = "";

  if (file && file.size > 0) {
    const bytes = Buffer.from(await file.arrayBuffer());
    const uploadDir = path.join(process.cwd(), "public", "uploads", "bank-transfers");
    await mkdir(uploadDir, { recursive: true });

    const finalName = `${Date.now()}-${safeFileName(file.name || "justificatif.pdf")}`;
    const finalPath = path.join(uploadDir, finalName);

    await writeFile(finalPath, bytes);
    proof_url = `/uploads/bank-transfers/${finalName}`;
  }

  const supabase = getSupabaseAdmin();

  if (supabase) {
    await supabase.from("client_payments").insert({
      client_email,
      client_name,
      package_name,
      amount,
      currency: "USD",
      payment_method: "bank_transfer",
      payment_status: "pending_verification",
      status: "pending_verification",
      reference,
      proof_url,
      phone,
      notes,
      created_at: new Date().toISOString(),
      updated_at: new Date().toISOString(),
    });
  }

  const nextUrl =
    lang === "en"
      ? `/en/account-verification?payment=bank_transfer_pending&email=${encodeURIComponent(client_email)}`
      : `/fr/verification-compte?payment=virement_en_attente&email=${encodeURIComponent(client_email)}`;

  return NextResponse.redirect(new URL(nextUrl, request.url), 303);
}

// @ts-nocheck

import { NextResponse } from "next/server";

export const runtime = "nodejs";
export const dynamic = "force-dynamic";

export async function POST(request: Request) {
  const body = await request.json().catch(() => ({}));

  const amount = Math.max(1, Number(body.amount || 179));
  const packageName = String(body.package_name || "New Mexico Standard");
  const email = String(body.email || body.client_email || "").trim().toLowerCase();

  const secretKey = process.env.STRIPE_SECRET_KEY || "";
  const origin = process.env.NEXT_PUBLIC_SITE_URL || new URL(request.url).origin;

  if (!secretKey) {
    return NextResponse.json(
      {
        ok: false,
        message: "STRIPE_SECRET_KEY est manquant dans .env.local",
      },
      { status: 500 }
    );
  }

  const params = new URLSearchParams();
  params.set("mode", "payment");
  params.set("success_url", origin + "/fr/verification-compte?payment=stripe_success&session_id={CHECKOUT_SESSION_ID}");
  params.set("cancel_url", origin + "/fr/stripe?payment=cancelled");

  if (email.includes("@")) {
    params.set("customer_email", email);
  }

  params.set("line_items[0][quantity]", "1");
  params.set("line_items[0][price_data][currency]", "usd");
  params.set("line_items[0][price_data][unit_amount]", String(Math.round(amount * 100)));
  params.set("line_items[0][price_data][product_data][name]", packageName);

  const response = await fetch("https://api.stripe.com/v1/checkout/sessions", {
    method: "POST",
    headers: {
      Authorization: "Bearer " + secretKey,
      "Content-Type": "application/x-www-form-urlencoded",
    },
    body: params,
  });

  const data = await response.json();

  if (!response.ok) {
    return NextResponse.json(
      {
        ok: false,
        message: data?.error?.message || "Erreur Stripe Checkout",
      },
      { status: 400 }
    );
  }

  return NextResponse.json({
    ok: true,
    id: data.id,
    url: data.url,
  });
}
